**Goal:** Build a Python MCP server that exposes surf forecast tools (get_surf_forecast, compare_spots, find_best_window) powered by Open-Meteo and NOAA APIs.

**Architecture:** Standalone Python package `surf_mcp` using FastMCP. Four modules handle concerns: `spots.py` (spot registry), `location.py` (location resolution), `open_meteo.py` (marine/weather API), `noaa.py` (tides), `server.py` (tool definitions). All HTTP via `httpx` async client.

**Tech Stack:** Python 3.10+, FastMCP, httpx, pytest, pytest-asyncio

# surf_mcp/rating.py

_LABELS = {1: "Poor", 2: "Fair", 3: "Good", 4: "Great", 5: "Epic"}


def compute_rating(wave_height_m: float, wave_period_s: float, wind_speed_kmh: float) -> int:
    """Compute a 1-5 condition rating from wave and wind data."""
    score = wave_height_m * wave_period_s / max(wind_speed_kmh, 5)
    if score < 2:
        return 1
    elif score < 4:
        return 2
    elif score < 8:
        return 3
    elif score <= 15:
        return 4
    else:
        return 5


def rating_label(rating: int) -> str:
    """Map a 1-5 rating to its label."""
    return _LABELS[rating]
```

# surf_mcp/geo.py
import math


def haversine_km(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    """Haversine distance between two points in km."""
    R = 6371.0
    dlat = math.radians(lat2 - lat1)
    dlon = math.radians(lon2 - lon1)
    a = math.sin(dlat / 2) ** 2 + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * math.sin(dlon / 2) ** 2
    return R * 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))
```

# surf_mcp/open_meteo.py
import httpx

_MARINE_URL = "https://marine-api.open-meteo.com/v1/marine"
_WEATHER_URL = "https://api.open-meteo.com/v1/forecast"
_GEOCODE_URL = "https://geocoding-api.open-meteo.com/v1/search"


class OpenMeteoError(Exception):
    pass


async def fetch_marine(lat: float, lon: float) -> dict:
    """Fetch 7-day hourly marine forecast from Open-Meteo."""
    params = {
        "latitude": lat,
        "longitude": lon,
        "hourly": "wave_height,wave_period,wave_direction",
        "forecast_days": 7,
        "timezone": "auto",
    }
    async with httpx.AsyncClient() as client:
        resp = await client.get(_MARINE_URL, params=params, timeout=10)
    if resp.status_code != 200:
        raise OpenMeteoError("Open-Meteo marine API unreachable. Try again shortly.")
    return resp.json()


async def fetch_weather(lat: float, lon: float) -> dict:
    """Fetch 7-day hourly wind + daily sunrise/sunset from Open-Meteo."""
    params = {
        "latitude": lat,
        "longitude": lon,
        "hourly": "wind_speed_10m,wind_direction_10m",
        "daily": "sunrise,sunset",
        "forecast_days": 7,
        "timezone": "auto",
    }
    async with httpx.AsyncClient() as client:
        resp = await client.get(_WEATHER_URL, params=params, timeout=10)
    if resp.status_code != 200:
        raise OpenMeteoError("Open-Meteo weather API unreachable. Try again shortly.")
    return resp.json()


async def geocode(name: str) -> tuple[float, float] | None:
    """Resolve a place name to (lat, lon) via Open-Meteo geocoding. Returns None if not found."""
    params = {"name": name, "count": 1, "language": "en"}
    async with httpx.AsyncClient() as client:
        resp = await client.get(_GEOCODE_URL, params=params, timeout=10)
    if resp.status_code != 200:
        return None
    data = resp.json()
    results = data.get("results")
    if not results:
        return None
    return (results[0]["latitude"], results[0]["longitude"])
```

# surf_mcp/noaa.py
import httpx

from surf_mcp.geo import haversine_km

_STATIONS_URL = "https://tidesandcurrents.noaa.gov/mdapi/latest/webapi/stations.json"
_PREDICTIONS_URL = "https://tidesandcurrents.noaa.gov/api/datagetter"
_MAX_STATION_DISTANCE_KM = 50

# In-memory cache for station list
_station_cache: list[dict] | None = None


class NoaaError(Exception):
    pass


async def _fetch_stations() -> list[dict]:
    """Fetch and cache the NOAA tide prediction station list."""
    global _station_cache
    if _station_cache is not None:
        return _station_cache
    async with httpx.AsyncClient() as client:
        resp = await client.get(_STATIONS_URL, params={"type": "tidepredictions"}, timeout=15)
    if resp.status_code != 200:
        return []
    _station_cache = resp.json().get("stations", [])
    return _station_cache


async def find_nearest_station(lat: float, lon: float) -> dict | None:
    """Find the nearest NOAA tide station within 50km. Returns None if none found."""
    stations = await _fetch_stations()
    best = None
    best_dist = float("inf")
    for s in stations:
        d = haversine_km(lat, lon, s["lat"], s["lng"])
        if d < best_dist:
            best_dist = d
            best = s
    if best is None or best_dist > _MAX_STATION_DISTANCE_KM:
        return None
    return best


async def fetch_tide_predictions(station_id: str, begin_date: str, end_date: str) -> dict[str, float]:
    """Fetch hourly tide predictions. Returns {timestamp_str: height_m} dict. Empty dict on error."""
    params = {
        "station": station_id,
        "product": "predictions",
        "datum": "MLLW",
        "units": "metric",
        "time_zone": "lst_ldt",
        "format": "json",
        "begin_date": begin_date,
        "end_date": end_date,
        "interval": "h",

# surf_mcp/location.py
import re
from dataclasses import dataclass

import httpx

from surf_mcp.spots import lookup_spot, SPOTS
from surf_mcp.open_meteo import geocode
from surf_mcp.geo import haversine_km

_IP_API_URL = "http://ip-api.com/json/"
_MAX_COAST_DISTANCE_KM = 50

# Simple regex for "lat,lon" format
_COORD_RE = re.compile(r"^(-?\d+\.?\d*)\s*,\s*(-?\d+\.?\d*)$")


class LocationError(Exception):
    pass


@dataclass
class ResolvedLocation:
    name: str
    lat: float
    lon: float
    note: str | None = None


def _nearest_spot(lat: float, lon: float) -> tuple[str, float, float, float]:
    """Find the nearest spot in the registry. Returns (name, lat, lon, distance_km)."""
    best_name = ""
    best_lat = 0.0
    best_lon = 0.0
    best_dist = float("inf")
    for name, (slat, slon) in SPOTS.items():
        d = haversine_km(lat, lon, slat, slon)
        if d < best_dist:
            best_dist = d
            best_name = name
            best_lat = slat
            best_lon = slon
    return best_name, best_lat, best_lon, best_dist


async def resolve_location(location: str) -> ResolvedLocation:
    """Resolve a location string to coordinates.

    Handles: "auto", known spot names, "lat,lon" coords, unknown names via geocoding.
    """
    location = location.strip()

    # Auto-detect via IP
    if location.lower() in ("auto", ""):
        return await _resolve_auto()

    # Check spot registry
    coords = lookup_spot(location)
    if coords is not None:
        # Title-case the name for display
        display_name = location.strip().title()
        return ResolvedLocation(name=display_name, lat=coords[0], lon=coords[1])

    # Check for raw coordinates
    match = _COORD_RE.match(location)
    if match:
        lat, lon = float(match.group(1)), float(match.group(2))
        return ResolvedLocation(name=f"{lat}, {lon}", lat=lat, lon=lon)

    # Geocode unknown name
    result = await geocode(location)
    if result is not None:
        lat, lon = result
        return ResolvedLocation(name=location.strip().title(), lat=lat, lon=lon)

    raise LocationError(
        f"Could not find location '{location}'. Try coordinates (lat,lon) or a known spot name."
    )


async def _resolve_auto() -> ResolvedLocation:
    """Resolve location via IP geolocation, snap to nearest coast if needed."""
    try:
        async with httpx.AsyncClient() as client:
            resp = await client.get(_IP_API_URL, timeout=5)
        if resp.status_code != 200:
            raise LocationError("Could not auto-detect location. Please specify a spot name or coordinates.")
        data = resp.json()
        lat, lon = data["lat"], data["lon"]
        city = data.get("city", "Unknown")
    except (httpx.HTTPError, KeyError):
        raise LocationError("Could not auto-detect location. Please specify a spot name or coordinates.")

    # Check if near coast — snap to nearest spot if far
    name, slat, slon, dist = _nearest_spot(lat, lon)
    if dist > _MAX_COAST_DISTANCE_KM:
        return ResolvedLocation(
            name=name.title(),
            lat=slat,
            lon=slon,
            note=f"Auto-detected location {city} is far from coast. Showing nearest spot: {name.title()}.",
        )
    return ResolvedLocation(name=city, lat=lat, lon=lon)
```

# surf_mcp/server.py
import json
from collections import defaultdict
from datetime import datetime, timedelta

from fastmcp import FastMCP

from surf_mcp.location import resolve_location, LocationError
from surf_mcp.open_meteo import fetch_marine, fetch_weather, OpenMeteoError
from surf_mcp.noaa import find_nearest_station, fetch_tide_predictions
from surf_mcp.rating import compute_rating, rating_label

mcp = FastMCP("surf-forecast")


def _build_forecast(marine: dict, weather: dict, tides: dict[str, float]) -> list[dict]:
    """Combine marine, weather, and tide data into the forecast structure."""
    # Group hourly data by date
    times = marine["hourly"]["time"]
    wave_h = marine["hourly"]["wave_height"]
    wave_p = marine["hourly"]["wave_period"]
    wave_d = marine["hourly"]["wave_direction"]
    wind_s = weather["hourly"]["wind_speed_10m"]
    wind_d = weather["hourly"]["wind_direction_10m"]

    # Build sunrise/sunset lookup
    sun_dates = weather["daily"]["time"]
    sunrises = weather["daily"]["sunrise"]
    sunsets = weather["daily"]["sunset"]
    sun_lookup = {}
    for i, d in enumerate(sun_dates):
        sun_lookup[d] = {
            "sunrise": sunrises[i].split("T")[1][:5] if "T" in sunrises[i] else sunrises[i],
            "sunset": sunsets[i].split("T")[1][:5] if "T" in sunsets[i] else sunsets[i],
        }

    # Group by date
    days: dict[str, list[dict]] = defaultdict(list)
    for i, t in enumerate(times):
        date_str = t[:10]  # "2026-03-20"
        time_str = t[11:16]  # "06:00"

        # Match tide by timestamp (NOAA format: "2026-03-20 06:00")
        tide_key = f"{date_str} {time_str}"
        tide_val = tides.get(tide_key)

        wh = wave_h[i] if i < len(wave_h) else 0
        wp = wave_p[i] if i < len(wave_p) else 0
        ws = wind_s[i] if i < len(wind_s) else 0

        rating = compute_rating(wh, wp, ws)

        days[date_str].append({
            "time": time_str,
            "wave_height_m": wh,
            "wave_period_s": wp,
            "wave_direction_deg": wave_d[i] if i < len(wave_d) else 0,
            "wind_speed_kmh": ws,
            "wind_direction_deg": wind_d[i] if i < len(wind_d) else 0,
            "tide_height_m": tide_val,
            "condition_rating": rating,
        })

    forecast = []
    for date_str in sorted(days.keys()):
        sun = sun_lookup.get(date_str, {"sunrise": "", "sunset": ""})
        forecast.append({
            "date": date_str,
            "sunrise": sun["sunrise"],
            "sunset": sun["sunset"],
            "hours": days[date_str],
        })
    return forecast


@mcp.tool()
async def get_surf_forecast(location: str = "auto") -> str:
    """Get a 7-day hourly surf forecast for a location.

    Args:
        location: Spot name (e.g., "Mavericks"), coordinates ("lat,lon"), or "auto" for auto-detect.
    """
    try:
        loc = await resolve_location(location)
    except LocationError as e:
        return json.dumps({"error": str(e)})

    try:
        marine = await fetch_marine(loc.lat, loc.lon)
        weather = await fetch_weather(loc.lat, loc.lon)
    except OpenMeteoError as e:
        return json.dumps({"error": str(e)})

    # Check for landlocked coords (no marine data)
    hourly = marine.get("hourly", {})
    wave_heights = hourly.get("wave_height", [])
    if not wave_heights or all(v is None for v in wave_heights):
        return json.dumps({"error": "No marine data available for this location."})

    # Tides
    tide_source = None
    tide_note = None
    tides: dict[str, float] = {}

    station = await find_nearest_station(loc.lat, loc.lon)
    if station:
        today = datetime.now().strftime("%Y%m%d")
        end = (datetime.now() + timedelta(days=7)).strftime("%Y%m%d")
        tides = await fetch_tide_predictions(station["id"], today, end)
        if tides:
            tide_source = f"NOAA CO-OPS (station {station['id']}, {station['name']})"
        else:
            tide_note = "Tide data temporarily unavailable."
    else:
        tide_note = "Tide data unavailable for this location (non-US)."

    forecast = _build_forecast(marine, weather, tides)

    result: dict = {
        "spot": loc.name,
        "coordinates": [loc.lat, loc.lon],
        "timezone": marine.get("timezone", "UTC"),
    }
    if tide_source:
        result["tide_source"] = tide_source
    if tide_note:
        result["tide_note"] = tide_note
    if loc.note:
        result["location_note"] = loc.note
    result["forecast"] = forecast

    return json.dumps(result)
```

@mcp.tool()
async def compare_spots(locations: list[str], date: str = "") -> str:
    """Compare surf conditions across multiple spots for a given day.

    Args:
        locations: List of 1-5 spot names or "lat,lon" coordinates.
        date: ISO 8601 date (e.g., "2026-03-22"). Defaults to today.
    """
    if not locations or len(locations) > 5:
        return json.dumps({"error": "Provide 1-5 locations."})

    # Validate/default date
    if not date:
        target_date = datetime.now().strftime("%Y-%m-%d")
    else:
        target_date = date
        try:
            parsed = datetime.strptime(target_date, "%Y-%m-%d")
            today = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
            if parsed < today or parsed > today + timedelta(days=7):
                return json.dumps({
                    "error": f"Date '{target_date}' is outside the 7-day forecast window. "
                             f"Valid range: {today.strftime('%Y-%m-%d')} to {(today + timedelta(days=7)).strftime('%Y-%m-%d')}."
                })
        except ValueError:
            return json.dumps({"error": f"Invalid date format '{date}'. Use ISO 8601 (YYYY-MM-DD)."})

    spots_data = []
    for loc_str in locations:
        # Get full forecast for each spot
        forecast_json = await get_surf_forecast(loc_str)
        forecast = json.loads(forecast_json)

        if "error" in forecast:
            spots_data.append({"spot": loc_str, "error": forecast["error"]})
            continue

        # Find the target date in the forecast
        day_data = None
        sunrise = ""
        sunset = ""
        for d in forecast.get("forecast", []):
            if d["date"] == target_date:
                day_data = d
                sunrise = d.get("sunrise", "")
                sunset = d.get("sunset", "")
                break

        if day_data is None:
            spots_data.append({"spot": forecast["spot"], "error": f"No data for {target_date}"})
            continue

        hours = day_data["hours"]
        if not hours:
            spots_data.append({"spot": forecast["spot"], "error": "No hourly data"})
            continue

        peak_wave = max(h["wave_height_m"] for h in hours)
        avg_wind = sum(h["wind_speed_kmh"] for h in hours) / len(hours)
        avg_rating_raw = sum(h["condition_rating"] for h in hours) / len(hours)
        avg_rating = round(avg_rating_raw)
        avg_rating = max(1, min(5, avg_rating))

        # Best hours: daylight hours with rating >= 3, or top 3
        best = []
        for h in hours:
            if sunrise and sunset and sunrise <= h["time"] <= sunset and h["condition_rating"] >= 3:
                best.append(h["time"])
        if not best:
            ranked = sorted(hours, key=lambda h: h["condition_rating"], reverse=True)
            best = sorted([h["time"] for h in ranked[:3]])

        spots_data.append({
            "spot": forecast["spot"],
            "coordinates": forecast["coordinates"],
            "timezone": forecast.get("timezone", "UTC"),
            "peak_wave_height_m": round(peak_wave, 1),
            "best_hours": best,
            "avg_wind_speed_kmh": round(avg_wind, 1),
            "avg_condition_rating": avg_rating,
            "rating_label": rating_label(avg_rating),
        })

    return json.dumps({"date": target_date, "spots": spots_data})
```

@mcp.tool()
async def find_best_window(
    locations: list[str],
    min_wave_height: float = 0,
    max_wind_speed: float = 0,
    date: str = "",
) -> str:
    """Find the best surf windows across spots over the next 7 days.

    Args:
        locations: List of spot names or "lat,lon" coordinates.
        min_wave_height: Minimum wave height in meters (default: no minimum).
        max_wind_speed: Maximum wind speed in km/h (default: no maximum).
        date: Optional ISO 8601 date to constrain to a single day.
    """
    all_windows: list[dict] = []

    for loc_str in locations:
        forecast_json = await get_surf_forecast(loc_str)
        forecast = json.loads(forecast_json)

        if "error" in forecast:
            continue

        for day in forecast.get("forecast", []):
            if date and day["date"] != date:
                continue

            # Find contiguous blocks where thresholds are met (with 1-hour gap tolerance)
            hours = day["hours"]
            current_window: list[dict] = []
            gap_count = 0  # Track consecutive non-qualifying hours

            for h in hours:
                meets = True
                if min_wave_height > 0 and h["wave_height_m"] < min_wave_height:
                    meets = False
                if max_wind_speed > 0 and h["wind_speed_kmh"] > max_wind_speed:
                    meets = False

                if meets:
                    if gap_count == 1 and current_window:
                        # Merge across 1-hour gap — include the gap hour
                        current_window.append(h)
                    else:
                        if gap_count > 1 and current_window:
                            # Gap too large, flush previous window
                            if len(current_window) >= 2:
                                all_windows.append(_make_window(
                                    forecast["spot"], day["date"],
                                    forecast.get("timezone", "UTC"), current_window
                                ))
                            current_window = []
                        current_window.append(h)
                    gap_count = 0
                else:
                    if current_window:
                        gap_count += 1
                        if gap_count == 1:
                            # Tolerate 1-hour gap — include gap hour in window tentatively
                            current_window.append(h)
                        elif gap_count > 1:
                            # Gap exceeded — flush window (remove trailing gap hour)
                            current_window = current_window[:-1] if gap_count == 2 else current_window
                            if len(current_window) >= 2:
                                all_windows.append(_make_window(
                                    forecast["spot"], day["date"],
                                    forecast.get("timezone", "UTC"), current_window
                                ))
                            current_window = []
                            gap_count = 0
                    # else: not in a window, skip

            # Don't forget the last window
            # Remove trailing gap hour if present
            if current_window and gap_count == 1:
                current_window.pop()
            if len(current_window) >= 2:
                all_windows.append(_make_window(
                    forecast["spot"], day["date"],
                    forecast.get("timezone", "UTC"), current_window
                ))

    # Rank by avg condition rating descending, take top 10
    all_windows.sort(key=lambda w: w["avg_condition_rating"], reverse=True)
    top = all_windows[:10]
    for i, w in enumerate(top):
        w["rank"] = i + 1

    message = None
    if not top:
        message = "No windows matching your criteria found in the 7-day forecast."

    filters: dict = {
        "min_wave_height_m": min_wave_height,
        "max_wind_speed_kmh": max_wind_speed if max_wind_speed > 0 else None,
    }

    return json.dumps({"filters": filters, "windows": top, "message": message})


def _make_window(spot: str, date: str, timezone: str, hours: list[dict]) -> dict:
    """Build a window dict from a list of consecutive hour dicts."""
    return {
        "spot": spot,
        "date": date,
        "start_time": hours[0]["time"],
        "end_time": hours[-1]["time"],
        "duration_hours": len(hours),
        "timezone": timezone,
        "avg_wave_height_m": round(sum(h["wave_height_m"] for h in hours) / len(hours), 1),
        "avg_wave_period_s": round(sum(h["wave_period_s"] for h in hours) / len(hours), 1),
        "avg_wind_speed_kmh": round(sum(h["wind_speed_kmh"] for h in hours) / len(hours), 1),
        "avg_condition_rating": round(sum(h["condition_rating"] for h in hours) / len(hours)),
        "rating_label": rating_label(
            max(1, min(5, round(sum(h["condition_rating"] for h in hours) / len(hours))))
        ),
    }

# surf-mcp

MCP server for surf and marine weather forecasts. Powered by Open-Meteo and NOAA CO-OPS (free, no API keys).

## Tools

- **get_surf_forecast** — 7-day hourly forecast for any surf spot
- **compare_spots** — side-by-side comparison of up to 5 spots
- **find_best_window** — find optimal surf windows across spots and days

## Install

```bash
git clone <repo-url> && cd surf-mcp
python -m venv .venv && source .venv/bin/activate
pip install -e .
```

## Claude Code config

Add to your Claude Code settings:

```json
{
  "mcpServers": {
    "surf-forecast": {
      "command": "python",
      "args": ["-m", "surf_mcp"],
      "cwd": "/absolute/path/to/surf-mcp"
    }
  }
}
```

## Usage

Once configured, ask Claude things like:
- "What's the surf forecast at Mavericks this week?"
- "Compare Ocean Beach and Steamer Lane for Saturday"
- "Find me the best surf window this week with waves over 2m"

## Built-in spots

California: Mavericks, Ocean Beach, Pacifica, Steamer Lane, Pleasure Point, Santa Cruz, Huntington Beach, Trestles, Blacks Beach, Rincon, Malibu

Hawaii: Pipeline, Sunset Beach, Waimea Bay, Jaws

International: Nazaré, Hossegor, Jeffrey's Bay, Teahupo'o, Bells Beach

Or use any location by name (geocoded) or coordinates ("36.95,-122.02").
```
